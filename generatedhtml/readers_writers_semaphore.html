import threading import time import random import html from IPython.display import display, HTML # Number of philosophers and forks N = 4 # Four forks, each protected by a semaphore forks = [threading.Semaphore(1) for _ in range(N)] # Store philosopher states and fork ownership states = ["Thinking"] * N fork_owner = [None] * N events = [] lock = threading.Lock() def log_event(message): with lock: events.append({ "states": states.copy(), "forks": fork_owner.copy(), "message": message }) def philosopher(i): left = i right = (i + 1) % N for round_no in range(1, 3): with lock: states[i] = "Thinking" log_event(f"Philosopher {i} is thinking") time.sleep(random.uniform(0.1, 0.3)) with lock: states[i] = "Hungry" log_event(f"Philosopher {i} is hungry") # Acquire forks in numerical order to prevent deadlock first, second = sorted([left, right]) forks[first].acquire() with lock: fork_owner[first] = i states[i] = "Acquiring forks" log_event(f"Philosopher {i} acquired fork {first}") forks[second].acquire() with lock: fork_owner[second] = i states[i] = "Eating" log_event(f"Philosopher {i} acquired fork {second} and is eating") time.sleep(random.uniform(0.1, 0.3)) # Release both forks forks[second].release() with lock: fork_owner[second] = None log_event(f"Philosopher {i} released fork {second}") forks[first].release() with lock: fork_owner[first] = None states[i] = "Thinking" log_event(f"Philosopher {i} released fork {first}") log_event(f"Philosopher {i} finished") # Start all four philosophers concurrently threads = [ threading.Thread(target=philosopher, args=(i,)) for i in range(N) ] for t in threads: t.start() for t in threads: t.join() print("Dining Philosophers simulation completed.") print("Total execution events:", len(events)) #
